#!/bin/bash
set -eu
root=/home/maxsp/stormworks
export JAVA_HOME=/home/maxsp/.local/opt/jdk-21.0.12.1+1
/home/maxsp/.local/opt/ghidra_12.1.4_PUBLIC/support/analyzeHeadless "$root/investigation/native/followups/tools/project" followups -import "$root/stormworks64.exe" -noanalysis -scriptPath "$root/investigation/native/followups/tools" -postScript RetainAt.java "$root/investigation/native/followups/pseudocode" "$@"
import ghidra.app.script.GhidraScript;
import ghidra.app.decompiler.*;
import ghidra.program.model.address.*;
import ghidra.program.model.listing.*;
import ghidra.app.cmd.disassemble.DisassembleCommand;
import ghidra.app.cmd.function.CreateFunctionCmd;
import java.io.*;

public class RetainAt extends GhidraScript {
  public void run() throws Exception {
    String[] args = getScriptArgs();
    String outDir = args[0];
    DecompInterface di = new DecompInterface();
    di.openProgram(currentProgram);
    for (int i = 1; i < args.length; i++) {
      Address a = toAddr(args[i]);
      DisassembleCommand dc = new DisassembleCommand(a, null, true);
      dc.applyTo(currentProgram, monitor);
      Function f = getFunctionAt(a);
      if (f == null) {
        CreateFunctionCmd cf = new CreateFunctionCmd(a);
        cf.applyTo(currentProgram, monitor);
        f = getFunctionAt(a);
      }
      PrintWriter pw = new PrintWriter(new FileWriter(outDir + "/" + args[i] + ".c"));
      if (f == null) { pw.println("// no function"); pw.close(); continue; }
      DecompileResults r = di.decompileFunction(f, 120, monitor);
      if (r.decompileCompleted()) { pw.println("// entry " + f.getEntryPoint()); pw.println(r.getDecompiledFunction().getC()); }
      else pw.println("// failed: " + r.getErrorMessage());
      pw.close();
      PrintWriter aw = new PrintWriter(new FileWriter(outDir + "/" + args[i] + ".addresses"));
      if (r.decompileCompleted()) {
        java.util.Iterator<ghidra.program.model.pcode.PcodeOpAST> it = r.getHighFunction().getPcodeOps();
        java.util.TreeSet<String> addresses = new java.util.TreeSet<>();
        while (it.hasNext()) addresses.add(it.next().getSeqnum().getTarget().toString());
        for (String address : addresses) aw.println(address);
      }
      aw.close();
      println("done " + args[i]);
    }
  }
}
